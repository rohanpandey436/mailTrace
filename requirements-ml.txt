# Optional transformer backend for Stage 3A (DistilRoBERTa NLP intent).
#
# NOT installed by default and deliberately NOT referenced from requirements.txt:
# torch alone is roughly 1 GB unpacked, so this does not fit the 512 MB free
# tier the public demo runs on. Without these packages MailTrace uses the
# bundled TF-IDF + logistic-regression classifier, which carries exact SHAP
# attributions and scores 0.90 accuracy on a stratified hold-out.
#
# To enable the transformer locally:
#     pip install -r requirements-ml.txt
#     set MAILTRACE_TRANSFORMER_MODEL=<a sequence-classification model id>
#
# Any Hugging Face sequence-classification model works; the label mapping in
# app/ml/train.py understands the common phishing/benign/spam/scam spellings.
# The first run downloads the weights, so it needs network access and disk.
transformers>=4.44
torch>=2.2
